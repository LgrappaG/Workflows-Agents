## 4. Network Manager Setup

```csharp
using Unity.Netcode;
using Unity.Services.Core;
using Unity.Services.Relay;
using Unity.Services.Relay.Models;
using Unity.Networking.Transport.Relay;

public class NetworkSetup : MonoBehaviour
{
    [SerializeField] private int _maxPlayers = 4;

    private async void Start()
    {
        await UnityServices.InitializeAsync();
    }

    public async Task<string> StartHost()
    {
        // Create relay allocation
        Allocation allocation = await RelayService.Instance.CreateAllocationAsync(_maxPlayers);
        string joinCode = await RelayService.Instance.GetJoinCodeAsync(allocation.AllocationId);

        // Configure transport
        var relayServerData = new RelayServerData(allocation, "dtls");
        NetworkManager.Singleton.GetComponent<UnityTransport>().SetRelayServerData(relayServerData);

        // Start host
        NetworkManager.Singleton.StartHost();

        return joinCode;
    }

    public async Task JoinGame(string joinCode)
    {
        // Join relay allocation
        JoinAllocation allocation = await RelayService.Instance.JoinAllocationAsync(joinCode);

        // Configure transport
        var relayServerData = new RelayServerData(allocation, "dtls");
        NetworkManager.Singleton.GetComponent<UnityTransport>().SetRelayServerData(relayServerData);

        // Start client
        NetworkManager.Singleton.StartClient();
    }
}
```

## 5. Lobby System

```csharp
using Unity.Services.Lobbies;
using Unity.Services.Lobbies.Models;

public class LobbyManager : MonoBehaviour
{
    private Lobby _currentLobby;

    public async Task<Lobby> CreateLobby(string lobbyName, int maxPlayers)
    {
        var options = new CreateLobbyOptions
        {
            IsPrivate = false,
            Data = new Dictionary<string, DataObject>
            {
                { "GameMode", new DataObject(DataObject.VisibilityOptions.Public, "Survival") },
                { "RelayCode", new DataObject(DataObject.VisibilityOptions.Member, "") }
            }
        };

        _currentLobby = await LobbyService.Instance.CreateLobbyAsync(lobbyName, maxPlayers, options);

        // Start heartbeat to keep lobby alive
        StartCoroutine(HeartbeatLobby());

        return _currentLobby;
    }

    public async Task<List<Lobby>> FindLobbies()
    {
        var options = new QueryLobbiesOptions
        {
            Filters = new List<QueryFilter>
            {
                new QueryFilter(QueryFilter.FieldOptions.AvailableSlots, "0", QueryFilter.OpOptions.GT)
            }
        };

        var response = await LobbyService.Instance.QueryLobbiesAsync(options);
        return response.Results;
    }

    public async Task JoinLobby(string lobbyId)
    {
        _currentLobby = await LobbyService.Instance.JoinLobbyByIdAsync(lobbyId);
    }

    private IEnumerator HeartbeatLobby()
    {
        while (_currentLobby != null)
        {
            LobbyService.Instance.SendHeartbeatPingAsync(_currentLobby.Id);
            yield return new WaitForSeconds(15f);
        }
    }
}
```

## 6. Player Spawn System

```csharp
using Unity.Netcode;

public class PlayerSpawner : NetworkBehaviour
{
    [SerializeField] private GameObject _playerPrefab;
    [SerializeField] private Transform[] _spawnPoints;

    public override void OnNetworkSpawn()
    {
        if (IsServer)
        {
            NetworkManager.Singleton.OnClientConnectedCallback += OnClientConnected;
        }
    }

    private void OnClientConnected(ulong clientId)
    {
        // Find spawn point
        int spawnIndex = (int)(clientId % (ulong)_spawnPoints.Length);
        Transform spawnPoint = _spawnPoints[spawnIndex];

        // Spawn player
        GameObject player = Instantiate(_playerPrefab, spawnPoint.position, spawnPoint.rotation);
        player.GetComponent<NetworkObject>().SpawnAsPlayerObject(clientId);
    }
}
```

## 7. VR Player Sync

```csharp
using Unity.Netcode;

public class VRPlayerSync : NetworkBehaviour
{
    [SerializeField] private Transform _head;
    [SerializeField] private Transform _leftHand;
    [SerializeField] private Transform _rightHand;

    // Synced transforms
    private NetworkVariable<Vector3> _headPos = new();
    private NetworkVariable<Quaternion> _headRot = new();
    private NetworkVariable<Vector3> _leftHandPos = new();
    private NetworkVariable<Quaternion> _leftHandRot = new();
    private NetworkVariable<Vector3> _rightHandPos = new();
    private NetworkVariable<Quaternion> _rightHandRot = new();

    private void Update()
    {
        if (IsOwner)
        {
            // Send local transform to network
            UpdateTransformsServerRpc(
                _head.position, _head.rotation,
                _leftHand.position, _leftHand.rotation,
                _rightHand.position, _rightHand.rotation
            );
        }
        else
        {
            // Apply network transforms to visuals
            _head.SetPositionAndRotation(_headPos.Value, _headRot.Value);
            _leftHand.SetPositionAndRotation(_leftHandPos.Value, _leftHandRot.Value);
            _rightHand.SetPositionAndRotation(_rightHandPos.Value, _rightHandRot.Value);
        }
    }

    [ServerRpc]
    private void UpdateTransformsServerRpc(
        Vector3 headPos, Quaternion headRot,
        Vector3 leftPos, Quaternion leftRot,
        Vector3 rightPos, Quaternion rightRot)
    {
        _headPos.Value = headPos;
        _headRot.Value = headRot;
        _leftHandPos.Value = leftPos;
        _leftHandRot.Value = leftRot;
        _rightHandPos.Value = rightPos;
        _rightHandRot.Value = rightRot;
    }
}
```

## 10. Common Issues & Solutions

| Issue | Cause | Solution |
|-------|-------|----------|
| Players don't see each other | NetworkObject not spawned | Check SpawnAsPlayerObject() |
| Rubber-banding | No interpolation | Add lerp on remote players |
| High latency feel | Client waiting for server | Implement prediction |
| "Not server" errors | Client calling ServerRpc | Check IsServer/IsOwner |
